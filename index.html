function generateMonsterPicture() { const loadingState = document.getElementById('loadingState'); const placeholderState = document.getElementById('placeholderState'); const generatedImage = document.getElementById('generatedImage'); const imageActions = document.getElementById('imageActions'); const loadingSubtext = document.getElementById('loadingSubtext'); if (loadingState) loadingState.classList.remove('hidden'); if (placeholderState) placeholderState.classList.add('hidden'); if (generatedImage) generatedImage.classList.add('hidden'); let msgIdx = 0; if (loadingMsgInterval) clearInterval(loadingMsgInterval); loadingMsgInterval = setInterval(() => { msgIdx = (msgIdx + 1) % loadingMessages.length; if (loadingSubtext) loadingSubtext.innerText = loadingMessages[msgIdx]; }, 1200); // Clean and encode text prompt safely for web requests const cleanPrompt = encodeURIComponent(currentPromptText); const seed = Math.floor(Math.random() * 1000000); // Primary URL (Using standard Pollinations image path) const primaryUrl = `https://image.pollinations.ai/prompt/${cleanPrompt}?width=512&height=512&seed=${seed}&nologo=true`; // Direct REST Fallback URL const fallbackUrl = `https://gen.pollinations.ai/image/${cleanPrompt}?width=512&height=512&seed=${seed}`; const imgLoader = new Image(); const handleSuccess = (url) => { if (loadingMsgInterval) clearInterval(loadingMsgInterval); if (loadingState) loadingState.classList.add('hidden'); if (generatedImage) { generatedImage.src = url; generatedImage.classList.remove('hidden'); } if (imageActions) imageActions.classList.remove('hidden'); addToGallery(url, document.getElementById('monsterName')?.value || 'Monster'); showToast('Monster image created!', 'fa-wand-magic-sparkles'); }; imgLoader.onload = function() { handleSuccess(primaryUrl); }; imgLoader.onerror = function() { const fallbackLoader = new Image(); fallbackLoader.onload = function() { handleSuccess(fallbackUrl); }; fallbackLoader.onerror = function() { if (loadingMsgInterval) clearInterval(loadingMsgInterval); if (loadingState) loadingState.classList.add('hidden'); if (placeholderState) placeholderState.classList.remove('hidden'); showToast('Could not generate image. Please try again!', 'fa-circle-exclamation'); }; fallbackLoader.src = fallbackUrl; }; imgLoader.src = primaryUrl; }
